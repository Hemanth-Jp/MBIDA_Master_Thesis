% ------------------------------------------------------------------
% Screenshots: compile even before the PNG exists.
%   \Screenshot[width]{Images/Folder/File.png}
% If the file is missing, a grey placeholder box with the file name is
% drawn instead, so the document always builds.
% ------------------------------------------------------------------
\newcommand{\Screenshot}[2][\linewidth]{%
  \IfFileExists{#2}{%
    \fcolorbox{ManGrey!50}{white}{\includegraphics[width=\dimexpr#1-2\fboxsep-2\fboxrule\relax]{#2}}%
  }{%
    \begin{tikzpicture}
      \node[draw=ManGrey,dashed,fill=ManGreyLight,minimum width=#1,minimum height=0.45*#1,
            align=center,font=\sffamily\small,text=ManGrey]
            {\textbf{Screenshot missing}\\[2pt]\detokenize{#2}};
    \end{tikzpicture}%
  }%
}
